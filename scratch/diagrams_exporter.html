erDiagram
    USERS ||--o{ APPLICATIONS : "submits (1:N)"
    USERS ||--o{ MESSAGES : "sends (1:N)"
    USERS ||--o{ ACTIVITY_LOGS : "generates (1:N)"
    VACANCIES }o--|| STAFFING_POSITIONS : "links to (N:1)"
    VACANCIES ||--o{ APPLICATIONS : "receives (1:N)"
    APPLICATIONS ||--o{ MESSAGES : "contains (1:N)"
    APPLICATIONS ||--o{ INTERVIEWS : "schedules (1:N)"

    USERS {
        bigint id PK
        string name
        string email UK
        string password
        string role "super_admin | hr_admin | hr_staff | applicant"
        json profile_data
        timestamp two_factor_confirmed_at
    }

    VACANCIES {
        bigint id PK
        bigint staffing_id FK
        string title
        string department
        string employment_type "Full-time | Part-time | Contractual"
        string location
        string salary_grade
        text description
        json responsibilities
        json requirements
        date deadline
        string status "Open | Closed | Archived"
    }

    APPLICATIONS {
        bigint id PK
        bigint job_id FK
        string email FK
        string applicant_name
        string phone_number
        string education
        string status "Submitted | Under Review | Shortlisted | Interview Scheduled | Hired | Rejected"
        json to_follow_docs
        json custom_file_responses
        json dynamic_responses
    }

    MESSAGES {
        bigint id PK
        bigint application_id FK
        bigint sender_id FK
        text content
        boolean is_read
    }

    INTERVIEWS {
        bigint id PK
        bigint application_id FK
        datetime scheduled_at
        string location
        string status "Scheduled | Completed | Cancelled"
    }

    STAFFING_POSITIONS {
        bigint id PK
        string title
        string department
        string status "Filled | Unfilled"
    }

    ACTIVITY_LOGS {
        bigint id PK
        bigint user_id FK
        string action
        text description
    }
